# =============================================
# Comandos de Python
# =============================================
# crear un ambiente virtual
python3 -m venv .venv

# activar ambiente virtual
source .venv/bin/activate

# instalar librerias desde archivo "requirements"
pip3 install -r requirements.txt

# =============================================
# Comandos de Git
# =============================================
# revisar status
git status

# agregar archivo
git add "nombre archivo" | ej: git add main.py

# confirmar cambios
git commit -m "comentario" | ej de comentario: "backup"

# cargar cambios al ecosistema
git push origin main
